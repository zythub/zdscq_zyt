#### 树结构导入数据

```
module.exports = function (request, ctx, params) {
  let id = params?.currentNode?.id;
  const fileInput = document.createElement("input");
  fileInput.type = "file";
  fileInput.style.display = "none";
  fileInput.accept = ".xlsx,.xls";
  fileInput.addEventListener("change", (event) => {
    const file = event.target.files[0];
    if (!file) return;
    const formData = new FormData();
    formData.append("file", file);
    formData.append("importType", "SUB_TABLE");
    formData.append("menuId", "0ba4bda6e81c0abd0ba14160ae8a7a8a");//要改这里为菜单id
    formData.append("relationValue", id);
    formData.append("time", new Date().getTime());
    request("POST", "/system/excel/import", {
      headers: {
        "Content-Type": "application/x-www-form-urlencoded",
      },
    })(formData)
      .then((response) => {
        // 成功提示
        if (params?.message?.success) {
          params.message.success("导入成功");
        } else {
          alert("导入成功");
        }
        // 刷新列表
        if (typeof params?.reload === "function") {
          params.reload();
        }
      })
      .catch((error) => {
        console.error("导入失败:", error);
        if (params?.message?.error) {
          params.message.error(
            "导入失败：" + (error.message || "请检查文件格式"),
          );
        } else {
          alert("导入失败");
        }
      })
      .finally(() => {
        // 清理 DOM 元素
        if (fileInput.parentNode) {
          fileInput.parentNode.removeChild(fileInput);
        }
      });
  });

  // 触发文件选择
  document.body.appendChild(fileInput);
  fileInput.click();
};

#### 按钮选表单带数据添加到子表

```
module.exports = async function (params, ctx, block) {
  // 步骤1: 获取弹窗选中的数据
  const selected = params.instance.props.currentSelectItems || [];
  
  // 步骤2: 获取子表现有数据ID列表，用于去重
  const existingIds = ctx.getArrayValue("subtable_qi3yhq74d5hjshugc7").map(row => row.input_fn9rvtd4dg1bn0r5lm);
  
  // 步骤3: 获取子表操作API
  const arrayFormApi = ctx.getArrayFormApi("subtable_qi3yhq74d5hjshugc7");
  let hasDuplicates = false;

  // 步骤4: 遍历选中数据，去重后追加到子表
  if (selected.length > 0) {
    for (const item of selected) {
      if (!existingIds.includes(item.id)) {
        arrayFormApi.append({
          input_fn9rvtd4dg1bn0r5lm: item.id,
          input_azgeyz5kf4pyecrxwt: item.input_y4ft9j6yi4vepev9zh,
          textarea_gsqb90xockw76gdxqh: item.input_rpb9twgfvdnidb36d7
        });
      } else {
        hasDuplicates = true;
      }
    }
  }

  // 步骤5: 提示重复信息
  if (hasDuplicates) {
    ctx.message("不能重复新增，将跳过重复项!", "info");
  }

  // 步骤6: 延迟执行后续逻辑，确保子表数据已渲染
  setTimeout(async () => {
    try {
      // 步骤7: 获取完整的子表数据
      const allRows = ctx.getArrayValue("subtable_qi3yhq74d5hjshugc7") || [];
      if (allRows.length === 0) return;

      // 步骤8: 提取所有物料ID
      const materialIds = allRows
        .map(row => row.input_fn9rvtd4dg1bn0r5lm)
        .filter(id => id);
      if (materialIds.length === 0) return;

      // 步骤9: 可根据物料ID批量查询库存或其他数据（此处预留接口）
      
    } catch (error) {
      console.error("查询库存失败:", error);
    }
  }, 50);
};
