### 1.3 created 事件：按菜单类型显隐按钮

```js
module.exports = function (params, ctx, block) {
  // 通过 URL 区分发布后的菜单类型
  var href = decodeURIComponent(window.location.href);
  var type = href.split("customform/")[1];

  if (type == "MA_SpecialTools") {
    // 专用工器具
    if (ctx.getValue("input_koikxhgq5a2jlp6r0m") != null) {
      ctx.visibles("button_q13uxw2k4dv2x5vicj", true);
    }
  } else if (type == "MA_SpareParts") {
    // 备品备件
    if (ctx.getValue("input_koikxhgq5a2jlp6r0m") != null) {
      ctx.visibles("button_nuvvb3ui5rxpk17v14", true);
    }
  } else {
    // 设备材料出库
    if (ctx.getValue("input_koikxhgq5a2jlp6r0m") != null) {
      ctx.visibles("button_is8ix3fh8uqnv9r6zn", true);
    }
  }

  // 移动端/桌面端显隐
  const isMobile = ctx?.isMobile;
  ctx.setInstance("row_x06lv1wqzrp39ln1ak", "visible", isMobile);
  ctx.setInstance("blank_327td0gdngwyvnng47", "visible", !isMobile);

  // 延迟取子表数据查库存（示例为注释状态，按需打开）
  setTimeout(async () => {
    try {
      const allRows = ctx.getArrayValue("subtable_qi3yhq74d5hjshugc7") || [];
      if (allRows.length === 0) return;
      const materialIds = allRows
        .map((row) => row.input_fn9rvtd4dg1bn0r5lm)
        .filter((id) => id);
      if (materialIds.length === 0) return;
      // 此处可调用 /mis/resource/material/getAvailStockCount
    } catch (error) {
      console.error("查询库存失败:", error);
    }
  }, 50);
};
```

### 1.5 保存按钮：出库数量校验（注释版）

```js
module.exports = function (request, ctx, params) {
  const allRows = ctx.getArrayValue("subtable_qi3yhq74d5hjshugc7") || [];
  if (allRows.length === 0) {
    ctx.message("请添加出库明细", "warning");
    return false;
  }
  let hasError = false;
  allRows.forEach((row, index) => {
    const materialId = row.input_fn9rvtd4dg1bn0r5lm;
    const outboundQuantity = Number(row.number_wy4mvj3ki3mgoxn2ma) || 0;
    const availableStock = Number(row.input_y17w4mj4p9uoqagk86) || 0;
    if (!materialId) return;
    if (outboundQuantity === 0) {
      ctx.message(`第${index + 1}行: 出库数量不能为0，请手动删除`, "error");
      hasError = true;
      return;
    }
    if (outboundQuantity > availableStock) {
      ctx.message(
        `库存不足，第${index + 1}行: 出库数量为 ${outboundQuantity}，可用库存为 ${availableStock}`,
        "error",
      );
      hasError = true;
    }
  });
  if (hasError) return false;
  return true;
};
```

```js
module.exports = function (request, ctx, params) {
  const id = params?.currentNode?.id;
```

### 2.11 通过数据库字段找控件 id

```js
function getControlIdByField(formInstance, fieldName) {
  const stack = [formInstance];
  while (stack.length > 0) {
    const node = stack.pop();
    if (node.props?.dataField === fieldName && node.id) return node.id;
    if (node.children) for (const c of node.children) stack.push(c);
  }
  return undefined;
}

// 用法
```

### 2.12 自定义接口 + 存储过程：按表名生成编号

```sql
DELIMITER $$
DROP PROCEDURE IF EXISTS generate_project_code$$
CREATE PROCEDURE generate_project_code(
    IN p_table_name VARCHAR(64),
    IN p_xmbh VARCHAR(50),
    IN p_code_column VARCHAR(64),
    IN p_fixed_name VARCHAR(100)
)
BEGIN
    SET @sql = CONCAT(
        'SELECT CONCAT(',
        'LPAD(COALESCE(MAX(CAST(SUBSTRING(', p_code_column, ' FROM 1 FOR 3) AS UNSIGNED)), 0) + 1, 3, ''0''), ',
        '''-', p_fixed_name, ''' ',
        ') AS new_code ',
        'FROM ', p_table_name, ' ',
        'WHERE xmbh = ''', p_xmbh, ''' ',
        '  AND sys_deleteflag = 0'
    );
    PREPARE stmt FROM @sql;
    EXECUTE stmt;
    DEALLOCATE PREPARE stmt;
END$$
DELIMITER ;
```

## 3. 实战要点

- **不要硬编码控件 id**：测试/正式环境控件 id 可能不同，优先用 `dataBind` + `getControlIdByField`。
- **保存表单用 PUT `/apis/dynamic/form`**：不要用 `/form/{id}`（405），也不要用 `/form/attribute`（dyFormFields 会变 null）。
- **按钮 clickEvent 可能 URL 编码**：读取时先 `decodeURIComponent`，保存时再 `encodeURIComponent`。
- **一个表单发布多个菜单**：用 URL `customform/{code}` 区分，created 事件里按需显隐字段/按钮。
- **子表操作一律走 ArrayFormApi**：不要直接 `ctx.setArrayValue` 追加新行（会缺内部状态），新增用 `api.append()`。
